import 'package:flutter/material.dart';

void main() { runApp(MaterialApp(home: FruitApp())); }
class FruitApp extends StatelessWidget {
  final fruits = {'Apple': 120, 'Banana': 50, 'Mango': 100};
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Fruit Market')),
      body: ListView(children: fruits.entries.map((fruit) {
        return FruitCard(name: fruit.key, price: fruit.value);
      }).toList()),
    );
  }
}
class FruitCard extends StatelessWidget {
  final String name;
  final int price;
  const FruitCard({super.key, required this.name, required this.price});
  @override
  Widget build(BuildContext context) {
    return Card(child: ListTile(
      title: Text(name), subtitle: Text('Rs. $price per kg'),
    ));
  }
}

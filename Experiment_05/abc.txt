import 'package:flutter/material.dart';

void main() {
  runApp(MaterialApp(routes: {
    '/': (_) => Home(), '/details': (_) => Details(),
    '/feedback': (_) => FeedbackPage(),
  }));
}
class Home extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Courses')),
      body: ListView(children: ['Flutter', 'Python'].map((course) {
        return ListTile(title: Text(course), onTap: () {
          Navigator.pushNamed(context, '/details', arguments: course);
        });
      }).toList()),
    );
  }
}
class Details extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final course = (ModalRoute.of(context)?.settings.arguments as String?) ?? 'Course';
    return Scaffold(
      appBar: AppBar(title: Text(course)),
      body: Center(child: ElevatedButton(
        child: const Text('Give Feedback'),
        onPressed: () async {
          final result = await Navigator.pushNamed(context, '/feedback');
          if (!context.mounted || result is! String) return;
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text(result)));
        },
      )),
    );
  }
}
class FeedbackPage extends StatefulWidget {
  @override
  State<FeedbackPage> createState() => _FeedbackPageState();
}
class _FeedbackPageState extends State<FeedbackPage> {
  String comment = '';
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Feedback')),
      body: Column(children: [
        TextField(onChanged: (v) => comment = v,
          decoration: const InputDecoration(labelText: 'Feedback')),
        ElevatedButton(child: const Text('Submit'), onPressed: () {
          if (comment.trim().isEmpty) {
            ScaffoldMessenger.of(context).showSnackBar(
              const SnackBar(content: Text('Enter feedback')));
          } else {
            Navigator.pop(context, comment.trim());
          }
        }),
      ]),
    );
  }
}

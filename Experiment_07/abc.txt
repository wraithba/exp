// Firestore employee add, duplicate-ID check, live list and delete. Only ID and name are used.
// This is a separate study example, not the complete lib/main.dart.
// Requires firebase_options.dart from FlutterFire and Firebase setup.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'firebase_options.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform);
  runApp(MaterialApp(home: Lab()));
}
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  String message = '';
  Future<void> perform(Future<void> Function() action) async {
    try {
      await action();
      if (mounted) setState(() => message = 'Operation completed');
    } catch (e) {
      if (mounted) setState(() => message = 'Error: $e');
    }
  }
  final employees = FirebaseFirestore.instance.collection('employees');
  String id = '', name = '';
  void save() {
    if (id.isEmpty || name.isEmpty || id.contains('/')) {
      setState(() => message = 'Enter valid ID and name'); return;
    }
    final ref = employees.doc(id);
    final enteredName = name;
    perform(() => FirebaseFirestore.instance.runTransaction<void>((tx) async {
      final existing = await tx.get(ref);
      if (existing.exists) throw Exception('Duplicate employee ID');
      tx.set(ref, {'name': enteredName});
    }));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Employees')),
      body: Column(children: [
        TextField(decoration: const InputDecoration(labelText: 'Employee ID'),
          onChanged: (v) => id = v.trim()),
        TextField(decoration: const InputDecoration(labelText: 'Name'),
          onChanged: (v) => name = v.trim()),
        ElevatedButton(onPressed: save, child: const Text('Add employee')),
        Text(message),
        Expanded(child: StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
          stream: employees.snapshots(), builder: (context, snapshot) {
            if (snapshot.hasError) return Text('Error: ${snapshot.error}');
            if (!snapshot.hasData) return const Text('Loading...');
            return ListView(children: snapshot.data!.docs.map((doc) {
              return ListTile(title: Text('${doc.data()['name']}'),
                subtitle: Text(doc.id), trailing: IconButton(
                  icon: const Icon(Icons.delete),
                  onPressed: () => perform(() => doc.reference.delete())));
            }).toList());
          })),
      ]),
    );
  }
}

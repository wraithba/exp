// Case-insensitive name search and maximum-price filter. Category, stock, sorting and reset are omitted.
// This is a separate study example, not the complete lib/main.dart.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';

void main() { runApp(MaterialApp(home: Lab())); }
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  String search = '';
  double limit = 3000;
  final products = {'Watch': 2000, 'Shirt': 800, 'Headphones': 1500};


  @override
  Widget build(BuildContext context) {
    final results = products.entries.where((p) =>
      p.key.toLowerCase().contains(search.toLowerCase()) && p.value <= limit);
    return Scaffold(
      appBar: AppBar(title: const Text('Search and Filter')),
      body: Column(children: [
        TextField(decoration: const InputDecoration(labelText: 'Search'),
          onChanged: (v) => setState(() => search = v)),
        Text('Maximum price: Rs. ${limit.toInt()}'),
        Slider(value: limit, min: 0, max: 3000, divisions: 30,
          onChanged: (v) => setState(() => limit = v)),
        Expanded(child: results.isEmpty
          ? const Center(child: Text('No products found'))
          : ListView(children: results.map((p) => ListTile(
              title: Text(p.key), subtitle: Text('Rs. ${p.value}'),
            )).toList())),
      ]),
    );
  }
}

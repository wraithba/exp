// Firebase Authentication registration/login/logout with a Firestore name/email profile. Confirmation, role and timestamps are omitted.
// This is a separate study example, not the complete lib/main.dart.
// Requires firebase_options.dart from FlutterFire and Firebase setup.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'firebase_options.dart';
import 'package:firebase_auth/firebase_auth.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform);
  runApp(MaterialApp(home: Lab()));
}
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  String message = '';
  Future<void> perform(Future<void> Function() action) async {
    try {
      await action();
      if (mounted) setState(() => message = 'Operation completed');
    } catch (e) {
      if (mounted) setState(() => message = 'Error: $e');
    }
  }
  final auth = FirebaseAuth.instance;
  String name = '', email = '', password = '';
  void register() {
    if (name.isEmpty || email.isEmpty || password.length < 6) {
      setState(() => message = 'Enter name, email and password (6+ characters)');
      return;
    }
    final enteredName = name, enteredEmail = email, enteredPassword = password;
    perform(() async {
      final credential = await auth.createUserWithEmailAndPassword(
        email: enteredEmail, password: enteredPassword);
      await FirebaseFirestore.instance.collection('users').doc(credential.user!.uid)
        .set({'name': enteredName, 'email': enteredEmail});
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('User Portal')),
      body: StreamBuilder<User?>(stream: auth.authStateChanges(),
        builder: (context, snapshot) {
          if (snapshot.hasError) return Text('Error: ${snapshot.error}');
          final user = snapshot.data;
          if (user != null) {
            return Column(children: [
              Text('Signed in: ${user.email}'),
              ElevatedButton(onPressed: () => perform(() => auth.signOut()),
                child: const Text('Logout')),
              Text(message),
            ]);
          }
          return SingleChildScrollView(child: Column(children: [
            TextField(decoration: const InputDecoration(labelText: 'Name'),
              onChanged: (v) => name = v.trim()),
            TextField(decoration: const InputDecoration(labelText: 'Email'),
              onChanged: (v) => email = v.trim()),
            TextField(obscureText: true,
              decoration: const InputDecoration(labelText: 'Password'),
              onChanged: (v) => password = v),
            ElevatedButton(onPressed: register, child: const Text('Register')),
            ElevatedButton(onPressed: () => perform(() async {
              await auth.signInWithEmailAndPassword(email: email, password: password);
            }), child: const Text('Login')),
            Text(message),
          ]));
        }),
    );
  }
}

import 'package:flutter/material.dart';

void main() => runApp(MaterialApp(home: Mileage()));

class Mileage extends StatefulWidget {
  @override
  State<Mileage> createState() => _MileageState();
}
class _MileageState extends State<Mileage> {
  String start = '', end = '', fuel = '', price = '', result = '';
  void calculate() {
    final a = double.tryParse(start), b = double.tryParse(end);
    final f = double.tryParse(fuel), p = double.tryParse(price);
    setState(() {
      if (a == null || b == null || f == null || p == null ||
          !a.isFinite || !b.isFinite || !f.isFinite || !p.isFinite ||
          a < 0 || b <= a || f <= 0 || p < 0) {
        result = 'Enter valid values';
      } else {
        result = 'Mileage: ${((b-a)/f).toStringAsFixed(2)} km/l\n'
            'Cost: Rs. ${(f*p).toStringAsFixed(2)}';
      }
    });
  }
  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: const Text('Mileage Calculator')),
    body: SingleChildScrollView(child: Column(children: [
      TextField(decoration: const InputDecoration(labelText: 'Initial km'),
        onChanged: (v) => start = v),
      TextField(decoration: const InputDecoration(labelText: 'Final km'),
        onChanged: (v) => end = v),
      TextField(decoration: const InputDecoration(labelText: 'Fuel litres'),
        onChanged: (v) => fuel = v),
      TextField(decoration: const InputDecoration(labelText: 'Price per litre'),
        onChanged: (v) => price = v),
      ElevatedButton(onPressed: calculate, child: const Text('Calculate')),
      Text(result),
    ])),
  );
}

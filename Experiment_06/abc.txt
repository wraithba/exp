import 'package:flutter/material.dart';
import 'package:sqflite_common_ffi/sqflite_ffi.dart';
import 'package:path/path.dart';

void main() {
  WidgetsFlutterBinding.ensureInitialized();
  sqfliteFfiInit();
  databaseFactory = databaseFactoryFfi;
  runApp(MaterialApp(home: Lab()));
}
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  Database? db;
  String roll = '', name = '', message = 'Opening database...';
  List<Map<String, Object?>> students = [];
  Future<void> load() async {
    final rows = await db!.query('students');
    if (mounted) setState(() { students = rows; });
  }
  Future<void> open() async {
    try {
      db = await openDatabase(join(await getDatabasesPath(), 'lab_students.db'),
        version: 1, onCreate: (db, version) async {
          await db.execute('CREATE TABLE students(roll TEXT PRIMARY KEY, name TEXT)');
        });
      await load();
      if (mounted) setState(() => message = 'Ready');
    } catch (e) {
      if (mounted) setState(() => message = 'Database error: $e');
    }
  }
  Future<void> save() async {
    if (roll.isEmpty || name.isEmpty) {
      setState(() => message = 'Enter roll number and name'); return;
    }
    try {
      await db!.insert('students', {'roll': roll, 'name': name});
      await load();
      if (mounted) setState(() => message = 'Saved');
    } catch (e) {
      if (mounted) setState(() => message = 'Save failed; check duplicate roll: $e');
    }
  }
  @override
  void initState() { super.initState(); open(); }
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('SQLite Students')),
      body: Column(children: [
        TextField(decoration: const InputDecoration(labelText: 'Roll number'),
          onChanged: (v) => roll = v.trim()),
        TextField(decoration: const InputDecoration(labelText: 'Name'),
          onChanged: (v) => name = v.trim()),
        ElevatedButton(onPressed: db == null ? null : save,
          child: const Text('Save student')),
        Text(message),
        Expanded(child: ListView(children: students.map((s) => ListTile(
          title: Text('${s['name']}'), subtitle: Text('${s['roll']}'),
        )).toList())),
      ]),
    );
  }
}

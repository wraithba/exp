// Firestore title-prefix and keyword search. Only titles are stored; author and genre are omitted.
// This is a separate study example, not the complete lib/main.dart.
// Requires firebase_options.dart from FlutterFire and Firebase setup.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'firebase_options.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform);
  runApp(MaterialApp(home: Lab()));
}
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  String message = '';
  Future<void> perform(Future<void> Function() action) async {
    try {
      await action();
      if (mounted) setState(() => message = 'Operation completed');
    } catch (e) {
      if (mounted) setState(() => message = 'Error: $e');
    }
  }
  final books = FirebaseFirestore.instance.collection('books');
  String title = '', search = '';
  bool keyword = false;
  List<QueryDocumentSnapshot<Map<String, dynamic>>> results = [];
  void find() {
    final text = search;
    perform(() async {
      Query<Map<String, dynamic>> query = books;
      if (text.isNotEmpty) {
        query = keyword ? books.where('keywords', arrayContains: text)
          : books.orderBy('titleLower').startAt([text]).endAt(['$text\uf8ff']);
      }
      final found = await query.get();
      if (mounted) setState(() => results = found.docs);
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Book Search')),
      body: Column(children: [
        TextField(decoration: const InputDecoration(labelText: 'New book title'),
          onChanged: (v) => title = v.trim()),
        ElevatedButton(onPressed: () {
          if (title.isEmpty) { setState(() => message = 'Enter a title'); return; }
          final text = title;
          perform(() async { await books.add({'title': text,
            'titleLower': text.toLowerCase(),
            'keywords': text.toLowerCase().split(RegExp(r'\s+'))}); });
        }, child: const Text('Add book')),
        TextField(decoration: const InputDecoration(labelText: 'Search text'),
          onChanged: (v) => search = v.trim().toLowerCase()),
        SwitchListTile(title: const Text('Keyword mode (off = prefix)'),
          value: keyword, onChanged: (v) => setState(() => keyword = v)),
        ElevatedButton(onPressed: find, child: const Text('Search')),
        Text(message),
        Expanded(child: ListView(children: results.map((doc) => ListTile(
          title: Text('${doc.data()['title']}'))).toList())),
      ]),
    );
  }
}

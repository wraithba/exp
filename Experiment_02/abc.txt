// Responsive grid using GridView.count. Category filters, prices and grid/list switching are omitted.
// This is a separate study example, not the complete lib/main.dart.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';

void main() { runApp(MaterialApp(home: ProductApp())); }
class ProductApp extends StatelessWidget {
  final products = ['Laptop', 'Phone', 'Watch', 'Shirt'];
  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Product Catalog')),
      body: LayoutBuilder(builder: (context, size) {
        return GridView.count(
          crossAxisCount: size.maxWidth < 600 ? 2 : 4,
          children: products.map((name) {
            return Card(child: Center(child: Text(name)));
          }).toList(),
        );
      }),
    );
  }
}

// Requires firebase_options.dart from FlutterFire and Firebase setup.
// Use the fuller program if your exam question asks for its extra features.

import 'package:flutter/material.dart';
import 'package:firebase_core/firebase_core.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'firebase_options.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp(options: DefaultFirebaseOptions.currentPlatform);
  runApp(MaterialApp(home: Lab()));
}
class Lab extends StatefulWidget {
  @override
  State<Lab> createState() => _LabState();
}
class _LabState extends State<Lab> {
  String message = '';
  Future<void> perform(Future<void> Function() action) async {
    try {
      await action();
      if (mounted) setState(() => message = 'Operation completed');
    } catch (e) {
      if (mounted) setState(() => message = 'Error: $e');
    }
  }
  final tasks = FirebaseFirestore.instance.collection('tasks');
  String title = '';


  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Task Updates')),
      body: Column(children: [
        TextField(decoration: const InputDecoration(labelText: 'Task title'),
          onChanged: (v) => title = v.trim()),
        ElevatedButton(child: const Text('Add task'), onPressed: () {
          if (title.isEmpty) { setState(() => message = 'Enter title'); return; }
          final text = title;
          perform(() async { await tasks.add({'title': text, 'done': false,
            'updatedAt': FieldValue.serverTimestamp()}); });
        }),
        Text(message),
        Expanded(child: StreamBuilder<QuerySnapshot<Map<String, dynamic>>>(
          stream: tasks.snapshots(), builder: (context, snapshot) {
            if (snapshot.hasError) return Text('Error: ${snapshot.error}');
            if (!snapshot.hasData) return const Text('Loading...');
            return ListView(children: snapshot.data!.docs.map((doc) {
              return CheckboxListTile(title: Text('${doc.data()['title']}'),
                value: doc.data()['done'] == true,
                onChanged: (value) => perform(() => doc.reference.update({
                  'done': value ?? false,
                  'updatedAt': FieldValue.serverTimestamp(),
                })));
            }).toList());
          })),
      ]),
    );
  }
}
